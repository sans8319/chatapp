import React, { useState, useEffect } from "react";
import "./OrderBookingModal.css"; 
import { 
  FiX, FiCalendar, FiMapPin, FiUsers, FiClipboard, FiCheckCircle, FiClock, FiPlus, FiChevronRight, FiChevronLeft
} from "react-icons/fi";
import { motion, AnimatePresence } from "framer-motion";
import axios from "axios";
import SubscriptionAlert from "./SubscriptionAlert"; 
import { SERVICE_MAPPING } from "./utils/serviceMapping";

const OrderBookingModal = ({ 
  categoryName, 
  serviceName, 
  mode,
  quantity,
  onClose, 
  isPremium, 
  onNavigateSub,
  preSelectedData
}) => {
    const stepIcons = {
      1: <FiCalendar />, 
      2: <FiMapPin />,   
      3: <FiUsers />,    
      4: <FiClipboard /> 
    };

  const [activeStep, setActiveStep] = useState(1);
  const [activeTab, setActiveTab] = useState("slots"); 
  const [showSubAlert, setShowSubAlert] = useState(false);
  const [loading, setLoading] = useState(false);
  
  // Data States from API
  const [addresses, setAddresses] = useState([]);
  const [members, setMembers] = useState([]);
  const [unitPrice, setUnitPrice] = useState(0);

  // Selection States
  const [selectedDate, setSelectedDate] = useState(preSelectedData?.date || "");
  const [selectedSlot, setSelectedSlot] = useState(preSelectedData?.time || "");
  const [selectedAddressId, setSelectedAddressId] = useState("");
  const [selectedMemberId, setSelectedMemberId] = useState("");
  const [customTime, setCustomTime] = useState("");

  const token = localStorage.getItem("token");

  // --- API Fetch Functions ---

  const fetchAddressesAndMembers = async () => {
    try {
      const headers = { Authorization: `Bearer ${token}` };
      const [addrRes, membRes] = await Promise.all([
        axios.get("/api/addresses", { headers }),
        axios.get("/api/members", { headers })
      ]);
      if (addrRes.data.success) setAddresses(addrRes.data.data);
      if (membRes.data.success) setMembers(membRes.data.data);
    } catch (err) {
      console.error("Error fetching addresses/members", err);
    }
  };

  const fetchPricing = async () => {
    setLoading(true);
    try {
      const mapping = SERVICE_MAPPING[serviceName]?.[mode];
      if (mapping) {
        const { subCategoryId, subSubCategoryId } = mapping;
        const res = await axios.get("/api/booking/get-unit-price", {
          params: { subId: subCategoryId, subSubId: subSubCategoryId },
          headers: { Authorization: `Bearer ${token}` }
        });
        if (res.data.success) {
          setUnitPrice(res.data.unitPrice);
        }
      }
    } catch (err) {
      console.error("Price fetch error", err);
    } finally {
      setLoading(false);
    }
  };

  // --- Navigation Logic ---

  const nextStep = () => {
    // Premium Check transition from Step 1 to 2
    if (activeStep === 1 && !isPremium) {
      setShowSubAlert(true);
      return;
    }

    // Load data when moving forward
    if (activeStep === 1) fetchAddressesAndMembers();
    if (activeStep === 3) fetchPricing();

    if (activeStep < 4) setActiveStep(prev => prev + 1);
  };

  const prevStep = () => setActiveStep(prev => prev - 1);

  // --- Steps Rendering ---

  const renderStep1 = () => (
    <div className="obm-step-content">
      <h4 className="obm-section-label">Select Date</h4>
      <div className="obm-date-wrapper">
        <FiCalendar className="obm-input-icon" />
        <input 
          type="date" 
          className="obm-date-input" 
          value={selectedDate}
          onChange={(e) => setSelectedDate(e.target.value)}
          min={new Date().toISOString().split("T")[0]}
        />
      </div>

      <h4 className="obm-section-label" style={{marginTop: '20px'}}>Preferred Time</h4>
      <div className="obm-tab-switcher">
        <button className={activeTab === "slots" ? "active" : ""} onClick={() => setActiveTab("slots")}>Available Slots</button>
        <button className={activeTab === "custom" ? "active" : ""} onClick={() => setActiveTab("custom")}>Custom Time</button>
      </div>

      {activeTab === "slots" ? (
        <div className="obm-slots-grid">
          {["09:00 AM", "12:00 PM", "03:00 PM", "06:00 PM"].map(slot => (
            <button 
              key={slot} 
              className={`obm-slot-btn ${selectedSlot === slot ? 'active' : ''}`}
              onClick={() => setSelectedSlot(slot)}
            >
              <FiClock /> {slot}
            </button>
          ))}
        </div>
      ) : (
        <div className="obm-date-wrapper">
           <FiClock className="obm-input-icon" />
           <input 
             type="time" 
             className="obm-date-input"
             value={customTime}
             onChange={(e) => {setCustomTime(e.target.value); setSelectedSlot(e.target.value)}}
           />
        </div>
      )}
    </div>
  );

  const renderStep2 = () => (
    <div className="obm-step-content">
      <div className="obm-section-header">
        <h4 className="obm-section-label">Select Address</h4>
        <button className="obm-add-btn"><FiPlus /> Add New</button>
      </div>
      <div className="obm-address-grid">
        {addresses.map(addr => (
          <div 
            key={addr.address_id} 
            className={`obm-address-card ${selectedAddressId === addr.address_id ? 'active' : ''}`}
            onClick={() => setSelectedAddressId(addr.address_id)}
          >
            <div className="addr-header">
              <span className="addr-tag">{addr.label}</span>
              {selectedAddressId === addr.address_id && <FiCheckCircle className="check-icon" />}
            </div>
            <p className="addr-line">{addr.address_line}</p>
            <p className="addr-city">{addr.city}, {addr.pin_code}</p>
          </div>
        ))}
      </div>
    </div>
  );

  const renderStep3 = () => (
    <div className="obm-step-content">
      <h4 className="obm-section-label">Who is this for?</h4>
      <div className="obm-member-list">
        {members.map(m => (
          <div 
            key={m.member_id} 
            className={`obm-member-item ${selectedMemberId === m.member_id ? 'selected' : ''}`}
            onClick={() => setSelectedMemberId(m.member_id)}
          >
            <div className="member-info">
                <div className="member-avatar"><FiUsers /></div>
                <div>
                  <p className="member-name">{m.name}</p>
                  <p className="member-relation">{m.phone || "No phone"}</p>
                </div>
            </div>
            {selectedMemberId === m.member_id && <FiCheckCircle className="selected-check" />}
          </div>
        ))}
      </div>
    </div>
  );

  const renderStep4 = () => (
    <div className="obm-step-content">
      {loading ? (
        <div className="obm-loading-shimmer">Calculating Final Price...</div>
      ) : (
        <div className="obm-summary-card">
          <div className="summary-item">
            <span>Service Details</span>
            <strong>{serviceName} ({mode})</strong>
          </div>
          <div className="summary-item">
            <span>Scheduled For</span>
            <strong>{selectedDate} at {selectedSlot}</strong>
          </div>
          <div className="summary-item">
            <span>Quantity</span>
            <strong>{quantity} Units</strong>
          </div>
          <div className="summary-item">
            <span>Unit Price</span>
            <strong>₹{unitPrice}</strong>
          </div>
          <div className="summary-divider"></div>
          <div className="summary-total">
            <span>Amount Payable</span>
            <span className="total-amount">₹{unitPrice * quantity}</span>
          </div>
        </div>
      )}
    </div>
  );

  return (
    <div className="obm-overlay">
      <motion.div 
        className="obm-modal"
        initial={{ y: "100%" }}
        animate={{ y: 0 }}
        exit={{ y: "100%" }}
        transition={{ type: "spring", damping: 25, stiffness: 200 }}
      >
        <div className="obm-header">
          <div className="obm-title-box">
            <FiCheckCircle className="obm-title-icon" />
            <div className="obm-title-text">
                <span className="obm-category-name">{categoryName}</span>
                Booking <span className="obm-highlight">{serviceName}</span>
                <span className="obm-mode-badge">{mode}</span>
            </div>
          </div>
          <button className="obm-close" onClick={onClose}><FiX /></button>
        </div>

        <div className="obm-progress-bar">
          {[1,2,3,4].map(step => (
            <div key={step} className={`obm-progress-step ${activeStep >= step ? 'active' : ''}`}>
              <div className="obm-dot">
                {activeStep > step ? <FiCheckCircle /> : stepIcons[step]}
              </div>
            </div>
          ))}
        </div>

        <div className="obm-body">
          <AnimatePresence mode="wait">
            <motion.div
              key={activeStep}
              initial={{ opacity: 0, x: 20 }}
              animate={{ opacity: 1, x: 0 }}
              exit={{ opacity: 0, x: -20 }}
            >
              {activeStep === 1 && renderStep1()}
              {activeStep === 2 && renderStep2()}
              {activeStep === 3 && renderStep3()}
              {activeStep === 4 && renderStep4()}
            </motion.div>
          </AnimatePresence>
        </div>

        <div className="obm-footer">
          {activeStep > 1 && <button className="obm-btn-back" onClick={prevStep}><FiChevronLeft /> Back</button>}
          <button 
            className="obm-btn-next" 
            onClick={activeStep === 4 ? () => {alert("Booking Placed Successfully!"); onClose();} : nextStep} 
            disabled={
                (activeStep === 1 && (!selectedDate || !selectedSlot)) || 
                (activeStep === 2 && !selectedAddressId) ||
                (activeStep === 3 && !selectedMemberId)
            }
          >
            {activeStep === 4 ? `Confirm & Pay ₹${unitPrice * quantity}` : 'Continue'} <FiChevronRight />
          </button>
        </div>
      </motion.div>

      <SubscriptionAlert 
        isOpen={showSubAlert} 
        onClose={() => setShowSubAlert(false)} 
        onConfirm={onNavigateSub} 
      />
    </div>
  );
};

export default OrderBookingModal;
